cmake_minimum_required(VERSION 3.24)
project(InputNextExamples LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "These examples require Windows and an MSVC-compatible toolchain.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(inputnext_core_example core/core_example.cpp core/core_reader.c)
add_executable(inputnext_effects_example effects/effects_example.cpp effects/effects_demo.c)
foreach(target IN ITEMS inputnext_core_example inputnext_effects_example)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../sdk")
    target_compile_features(${target} PRIVATE c_std_17 cxx_std_17)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endforeach()
